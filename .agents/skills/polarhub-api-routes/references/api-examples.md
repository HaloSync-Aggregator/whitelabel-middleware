# PolarHub API Examples

Implementation examples for each API.

## AirShopping

```typescript
// Workflow position: Starting point for all Prime Bookings
// [AirShopping] → OfferPrice → OrderCreate

interface AirShoppingQuery {
  OriginDestList: Array<{
    OriginCode: string;      // IATA airport code
    DestCode: string;
    DepartureDate: string;   // YYYY-MM-DD
  }>;
  PaxList: Array<{
    PaxID: string;           // PAX1, PAX2, ...
    Ptc: 'ADT' | 'CHD' | 'INF';
    BirthDate?: string;
    LoyaltyProgramAccount?: Array<{ ProgramCode: string; AccountNumber: string }>;
  }>;
  Cabin?: 'Y' | 'W' | 'C' | 'F';
  Criteria?: {
    Direct?: boolean;
    AirlinePreference?: { AirlineID: string[] };
  };
}

// Extract from response and pass to the next API
const { ResponseID, OfferID, Owner, OfferItem } = response.Offer[0];
```

```typescript
// Workflow position
// AirShopping → [SeatAvailability] → OfferPrice → OrderCreate (WF_PB_SEAT)
// OrderRetrieve → [SeatAvailability] → OrderChange (WF_HELD_SEAT)

  "Query": {
    "Offer": {

const selectedSeat = { Column: "G", Row: "20" };
```

## OrderCreate

```typescript
// Workflow position: Ending point for all Prime Bookings
// AirShopping → OfferPrice → [OrderCreate]

interface OrderCreateQuery {
  Order: Array<{
    ResponseID: string;
    OfferID: string;
    Owner: string;
    OfferItems: Array<{
      OfferItemID: string;
      PaxRefID: string[];
      SeatSelection?: { Column: string; Row: string };
      BookingInstructions?: { Text: string[]; Ositext: string[] };
    }>;
  }>;
  PaxList: PaxType[];
  PaymentList?: PaymentType[];      // Required for WF_PB_IMMEDIATE
  ContactInfoList?: ContactInfoType[];
}

// WF_PB_DEFERRED - PaymentList not required
// WF_PB_IMMEDIATE - PaymentList required
```

### Required ContactInfo Fields by Airline
```typescript
// BA - Email + PostalAddress required
{ ContactInfoID: "CTC_TA1", EmailAddress: [...], PostalAddress: [...] }

// AF, KL - Email + Phone required
{ ContactInfoID: "CTC_TA1", EmailAddress: [...], Phone: [...] }
```

## OrderRetrieve

```typescript
// Workflow position: Starting point for all Post-Bookings
// [OrderRetrieve] → SeatAvailability/ServiceList/OrderChange/OrderCancel/OrderReshop

// CRITICAL: TransactionID is generated by FE, middleware only passes through!
// ✅ FE code
const postBookingTxnId = crypto.randomUUID().replace(/-/g, '');  // Generated by FE
await fetch('/middleware/polarhub/order/retrieve', {
  body: JSON.stringify({ transactionId: postBookingTxnId, orderId: 'ORDER_ID_FROM_CREATE' })
});

// ❌ Must not be generated in middleware
const postBookingTxnId = generateTransactionId();  // Absolutely prohibited!
```

## OrderReshop

```typescript
// Workflow position
// OrderRetrieve → [OrderReshop] → OrderChange (WF_HELD_ITIN, WF_TKT_VOLCHANGE)
// OrderRetrieve → [OrderReshop] → OrderCancel (WF_TKT_REFUND)

// Itinerary change
{
  "Query": {
    "OrderID": "...",
    "ActionContext": "8",  // Voluntary
    "UpdateOrder": {
      "ReshopOrder": {
        "ServiceOrder": {
          "Delete": { "OrderItemRefID": ["OLD_SEGMENT"] },
          "Add": {
            "OriginDestList": [{ "OriginCode": "ICN", "DestCode": "SIN", "DepartureDate": "2025-03-01" }],
            "Cabin": "Y"
          }
        }
      }
    }
  }
}

// Refund
{
  "Query": {
    "OrderID": "...",
    "UpdateOrder": {
      "CancelOrderRefID": { "OrderID": "..." }
    }
  }
}
```
