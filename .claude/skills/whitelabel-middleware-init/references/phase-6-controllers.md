/**
 * AirShopping Controller
 *
 * Entry point for the PRE-BOOKING flow.
 *
 * CLAUDE.md principles:
 * - TransactionID: Generated by FE, pass-through
 * - Sender: FE-provided value takes priority, environment variable as Fallback
 * - All APIs use POST method
 * - Return raw response (no transformation)
 */

import {
  Controller,
  Post,
  Body,
  HttpCode,
  HttpStatus,
  Logger,
  Res,
} from '@nestjs/common';
import { Response } from 'express';
import {
  ApiTags,
  ApiOperation,
  ApiResponse,
  ApiBody,
  ApiHeader,
} from '@nestjs/swagger';
import { PolarHubService } from '../polarhub.service';
import {
  AirShoppingRequestDto,
  AirShoppingResponseDto,
  CriteriaDto,
} from '../dto/air-shopping.dto';

@ApiTags('air-shopping')
@Controller('polarhub/air-shopping')
export class AirShoppingController {
  private readonly logger = new Logger(AirShoppingController.name);

  constructor(private readonly polarHubService: PolarHubService) {}

  /**
   * Search for available flights
   *
   * First step of the PRE-BOOKING flow.
   * The returned TransactionID must be reused identically in OfferPrice and OrderCreate.
   */
  @Post()
  @HttpCode(HttpStatus.OK)
  @ApiOperation({
    summary: 'Search for available flights',
    description: `PRE-BOOKING flow starting point.

**TransactionID Rules:**
- TransactionID is generated by FE and passed through
- Same TransactionID must be used for subsequent OfferPrice and OrderCreate calls

**Trip Types:**
- One-way: 1 item in originDestList
- Round-trip: 2 items in originDestList
- Multi-city: N items in originDestList

**Passenger Rules:**
- Max 9 passengers total
- INF count must not exceed ADT count
- Passenger types: ADT (Adult 12+), CHD (Child 2-11), INF (Infant 0-1)

**Cabin Classes:**
- Y: Economy
- W: Premium Economy
- C: Business
- F: First

**Criteria Options:**
- direct: true for non-stop flights only
- airlinePreference: preferred airline IATA codes
- programCriteria: contract fare programs
- promotionCriteria: promotional fares
- specialFareCriteria: special fare programs`,
  })
  @ApiBody({ type: AirShoppingRequestDto })
  @ApiResponse({
    status: 200,
    description: 'Flight search results - returns raw PolarHub response',
    type: AirShoppingResponseDto,
  })
  @ApiResponse({
    status: 400,
    description: 'Invalid request (validation error)',
  })
  @ApiResponse({
    status: 401,
    description: 'Authentication failed (HMAC signature invalid)',
  })
  @ApiResponse({
    status: 500,
    description: 'Internal server error',
  })
  @ApiHeader({
    name: 'X-Transaction-Id',
    description: 'Transaction ID (returned in response header)',
    required: false,
  })
  async searchFlights(
    @Body() request: AirShoppingRequestDto,
    @Res({ passthrough: true }) res: Response,
  ): Promise<unknown> {
    // 1. Request logging
    this.logger.log({
      message: 'AirShopping request received',
      transactionId: request.transactionId,
      routes: request.originDestList.length,
      passengers: request.passengers.reduce((sum, p) => sum + p.count, 0),
      cabin: request.cabin,
      hasSender: !!request.sender,
      hasCriteria: !!request.criteria,
      hasLegacyFields: !!(request.airlinePreference || request.directOnly),
    });

    // 2. Criteria handling: full criteria object takes priority, legacy fields as fallback
    const effectiveAirlinePreference = this.getEffectiveAirlinePreference(request);
    const effectiveDirectOnly = this.getEffectiveDirectOnly(request);

    // 3. Call service
    const result = await this.polarHubService.searchFlights({
      // TransactionID pass-through (generated by FE)
      transactionId: request.transactionId,
      pointOfSale: request.pointOfSale,

      // Sender: FE-provided value takes priority, falls back to environment variables
      sender: request.sender
        ? {
            travelAgency: {
              agencyId: request.sender.travelAgency.agencyId,
              siteCode: request.sender.travelAgency.siteCode,
              agencyName: request.sender.travelAgency.agencyName,
              contactInfoRefId: request.sender.travelAgency.contactInfoRefId,
            },
          }
        : undefined,

      // FE Auth Headers pass-through
      feAuthHeaders: request.feAuthHeaders,

      // Query parameters
      originDestList: request.originDestList,
      passengers: request.passengers,
      cabin: request.cabin,
      // Full criteria object (pass-through all CriteriaType4 fields)
      criteria: request.criteria,
      // ResponseParameter
      responseParameter: request.responseParameter,
      // Legacy fields (fallback when criteria is not provided)
      airlinePreference: effectiveAirlinePreference,
      directOnly: effectiveDirectOnly,
    });

    // 4. Set response header
    res.setHeader('X-Transaction-Id', request.transactionId);

    // 5. Return raw response (no transformation!)
    return result.response;
  }

  /**
   * Extract airlinePreference from Criteria
   * criteria.airlinePreference takes priority, falls back to legacy airlinePreference
   */
  private getEffectiveAirlinePreference(
    request: AirShoppingRequestDto,
  ): string[] | undefined {
    // criteria.airlinePreference takes priority
    if (
      request.criteria?.airlinePreference?.airlineId &&
      request.criteria.airlinePreference.airlineId.length > 0
    ) {
      return request.criteria.airlinePreference.airlineId;
    }

    // Legacy field fallback
    if (request.airlinePreference && request.airlinePreference.length > 0) {
      this.logger.warn({
        message:
          'Using deprecated airlinePreference field. Use criteria.airlinePreference.airlineId instead.',
        transactionId: request.transactionId,
      });
      return request.airlinePreference;
    }

    return undefined;
  }

  /**
   * Extract direct flag from Criteria
   * criteria.direct takes priority, falls back to legacy directOnly
   */
  private getEffectiveDirectOnly(
    request: AirShoppingRequestDto,
  ): boolean | undefined {
    // criteria.direct takes priority
    if (request.criteria?.direct !== undefined) {
      return request.criteria.direct;
    }

    // Legacy field fallback
    if (request.directOnly !== undefined) {
      this.logger.warn({
        message:
          'Using deprecated directOnly field. Use criteria.direct instead.',
        transactionId: request.transactionId,
      });
      return request.directOnly;
    }

    return undefined;
  }
}

/**
 * Order Controller
 *
 * Unified controller for OrderCreate, OrderRetrieve, and OrderCancel APIs.
 *
 * CRITICAL:
 * - All APIs use POST method (PolarHub HTTP spec)
 * - transactionId must be provided by FE (middleware generation forbidden!)
 * - ContactInfo: Pass through as-is from FE (no data concatenation/modification)
 * - No hardcoding (CitizenshipCountryCode, etc.)
 *
 * ContactInfoRefID auto-injection (added 2026-01-21):
 * - sender.travelAgency.contactInfoRefId is forcefully injected from env var (POLARHUB_AGENCY_CONTACT_ID)
 * - If FE does not provide sender: Default sender is auto-generated based on env vars
 * - Even if FE provides sender: Only contactInfoRefId is forcefully replaced with env var value
 * - This ensures Travel Agency ContactInfo (CTC_TA1) is always included
 */

import {
  Controller,
  Post,
  Body,
  Logger,
  HttpCode,
  HttpStatus,
} from '@nestjs/common';
import {
  ApiTags,
  ApiOperation,
  ApiResponse,
  ApiBody,
} from '@nestjs/swagger';
import { PolarHubService } from '../polarhub.service';
import { OrderCreateRQDto } from '../dto/order-create.dto';
import { OrderRetrieveRQDto } from '../dto/order-retrieve.dto';
import { OrderCancelRQDto } from '../dto/order-cancel.dto';

@ApiTags('PolarHub - Order')
@Controller('polarhub/order')
export class OrderController {
  private readonly logger = new Logger(OrderController.name);

  constructor(private readonly polarHubService: PolarHubService) {}

  // ==============================================
  // OrderCreate - POST /middleware/polarhub/order
  // ==============================================

  @Post()
  @HttpCode(HttpStatus.OK)
  @ApiOperation({
    summary: 'Create Order (Booking)',
    description: `
## Overview
OrderCreate API is used to process flight booking and ticket issuance.

## Ticketing Modes
- **Deferred Ticketing**: Booking only, ticketing later (no PaymentList)
- **Instant Ticketing**: Booking and issuing immediately (with PaymentList)

## Form of Payment
- **BSP Cash**: All airlines (except TR)
- **BSP Credit Card**: HA, KE
- **Agent Deposit**: TR only

## Required ContactInfo
Some airlines require Travel Agency contact info:
- **BA**: Email, PostalAddress
- **AF/KL**: Email, Phone

## Important Notes
- TransactionID must be same as OfferPrice
- BA US routes: Passport & PostalAddress required
- HA/KE: IssuingCountryCode required in IdentityDoc
    `,
  })
  @ApiBody({ type: OrderCreateRQDto })
  @ApiResponse({
    status: 200,
    description: 'Order created successfully',
  })
  @ApiResponse({
    status: 400,
    description: 'Bad Request - Invalid request format',
  })
  @ApiResponse({
    status: 401,
    description: 'Unauthorized - HMAC authentication failed',
  })
  @ApiResponse({
    status: 500,
    description: 'Internal Server Error',
  })

    this.logger.log(
      `[OrderCreate] transactionId=${request.transactionId}, orders=${request.orders.length}, paxList=${request.paxList.length}`,
    );

    try {
      const { transactionId, response } = await this.polarHubService.createOrder({
        transactionId: request.transactionId,
        pointOfSale: request.pointOfSale,
        sender: request.sender
          ? {
              travelAgency: {
                agencyId: request.sender.travelAgency.agencyId,
                siteCode: request.sender.travelAgency.siteCode,
                agencyName: request.sender.travelAgency.agencyName,
                contactInfoRefId: request.sender.travelAgency.contactInfoRefId,
              },
            }
          : undefined,

        // Orders (Offer Selection)
        orders: request.orders.map((order) => ({
          responseId: order.responseId,
          offerId: order.offerId,
          owner: order.owner,
          offerItems: order.offerItems.map((item) => ({
            offerItemId: item.offerItemId,
            paxRefId: item.paxRefId,
            seatSelection: item.seatSelection,
            bookingInstructions: item.bookingInstructions,
          })),
        })),

        // Passengers
        paxList: request.paxList.map((pax) => ({
          paxId: pax.paxId,
          ptc: pax.ptc,
          individual: {
            nameTitle: pax.individual.nameTitle,
            givenName: pax.individual.givenName,
            surname: pax.individual.surname,
            birthdate: pax.individual.birthdate,
            gender: pax.individual.gender,
          },
          identityDoc: pax.identityDoc?.map((doc) => ({
            identityDocType: doc.identityDocType,
            identityDocId: doc.identityDocId,
            expiryDate: doc.expiryDate,
            issuingCountryCode: doc.issuingCountryCode,
            citizenshipCountryCode: doc.citizenshipCountryCode,
          })),
          contactInfoRefId: pax.contactInfoRefId,
          citizenshipCountryCode: pax.citizenshipCountryCode,
          residenceCountryCode: pax.residenceCountryCode,
        })),

        // ContactInfo (pass-through, no modification!)
        contactInfoList: request.contactInfoList.map((ci) => ({
          contactInfoId: ci.contactInfoId,
          emailAddress: ci.emailAddress,
          phone: ci.phone
            ? {
                phoneNumber: ci.phone.phoneNumber,
                countryDialingCode: ci.phone.countryDialingCode,
                label: ci.phone.label,
              }
            : undefined,
          postalAddress: ci.postalAddress
            ? {

      `ServiceList request - txnId: ${request.transactionId}, ` +
        `mode: ${request.offer ? 'Offer' : 'Order'}`,
    );

    // Compose service call parameters
    const params = {
      transactionId: request.transactionId,
      pointOfSale: request.pointOfSale,
      feAuthHeaders: request.feAuthHeaders,
      paxList: request.paxList.map((pax) => ({
        paxId: pax.paxId,
        ptc: pax.ptc,
        ...(pax.loyaltyProgramAccount && {
          loyaltyProgramAccount: pax.loyaltyProgramAccount.map((lpa) => ({
            accountNumber: lpa.accountNumber,
            programCode: lpa.programCode,
            programName: lpa.programName,
            signInId: lpa.signInId,
            tierCode: lpa.tierCode,
            tierName: lpa.tierName,
          })),
        }),
      })),
      // Sender: use FE-provided value if available, otherwise fall back to environment variables
      ...(request.sender && {
        sender: {
          travelAgency: {
            agencyId: request.sender.travelAgency.agencyId,
            siteCode: request.sender.travelAgency.siteCode,
            agencyName: request.sender.travelAgency.agencyName,
            contactInfoRefId: request.sender.travelAgency.contactInfoRefId,
          },
        },
      }),
      // Offer (Prime Booking)
      ...(request.offer && {
        offer: {
          responseId: request.offer.responseId,
          offerId: request.offer.offerId,
          owner: request.offer.owner,
          offerItems: request.offer.offerItems?.map((item) => ({
            offerItemId: item.offerItemId,
            paxRefId: item.paxRefId || [],
            ...(item.paxSegmentRefId && {
              paxSegmentRefId: item.paxSegmentRefId,
            }),
          })) || [],
        },
      }),
      // Order (Post-Booking)
      ...(request.order && {
        orderId: request.order.orderId,
        ...(request.order.orderItems && {
          orderItems: request.order.orderItems.map((item) => ({
            orderItemId: item.orderItemId,
            paxRefId: item.paxRefId,
          })),
        }),
      }),
      // Criteria (optional)
      ...(request.criteria && {
        criteria: {
          ...(request.criteria.programCriteria && {
            programCriteria: request.criteria.programCriteria.map((pc) => ({
              airlineId: pc.airlineId,
              contractId: pc.contractId,
              accountId: pc.accountId,
              typeCode: pc.typeCode,
            })),
          }),
          ...(request.criteria.specialFareCriteria && {
            specialFareCriteria: request.criteria.specialFareCriteria.map(
              (sfc) => ({
                airlineId: sfc.airlineId,
                accountId: sfc.accountId,
              }),
            ),
          }),
          ...(request.criteria.existingOrderCriteria && {
            existingOrderCriteria: {
              orderId: request.criteria.existingOrderCriteria.orderId,
              paxRefId: request.criteria.existingOrderCriteria.paxRefId,
            },
          }),
        },
      }),
      // ResponseParameter (optional)
      ...(request.responseParameter && {
        responseParameter: {
          currencyCode: request.responseParameter.currencyCode,
          ...(request.responseParameter.languageUsage && {
            languageUsage: {
              langCode: request.responseParameter.languageUsage.langCode,
              langUsageType: request.responseParameter.languageUsage.langUsageType,
            },
          }),
        },
      }),
    };

    // Call service

      `ServiceList response - txnId: ${result.transactionId}, ` +
        `hasServices: ${!!result.response?.ALaCarteOffer}`,
    );

    return {
      transactionId: result.transactionId,
      ...result.response,
    };
